# Compile the actual scalar aura-read functions with a small container fixture.
file(READ "${PROJECT_SOURCE_DIR}/src/game/Entities/Unit.cpp" _aura_unit)
string(FIND "${_aura_unit}" "int32 Unit::GetTotalAuraModifier(AuraType auratype) const" _aura_start)
string(FIND "${_aura_unit}" "bool Unit::AddSpellAuraHolder(" _aura_end)
math(EXPR _aura_length "${_aura_end}-${_aura_start}")
string(SUBSTRING "${_aura_unit}" ${_aura_start} ${_aura_length} _aura_functions)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/aura_read_functions.inc" "${_aura_functions}")
string(REGEX MATCHALL "(int32|float) Unit::[^\n]+" _aura_signatures "${_aura_functions}")
set(_aura_declarations "")
foreach(_aura_signature IN LISTS _aura_signatures)
  string(REPLACE "Unit::" "" _aura_signature "${_aura_signature}")
  string(APPEND _aura_declarations "${_aura_signature};\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/aura_read_declarations.inc" "${_aura_declarations}")
